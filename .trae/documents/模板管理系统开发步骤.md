def extract_page_setup(ws):

        "orientation": ws.page_setup.orientation,
        "margins": {
            "top": ws.page_margins.top,
            "right": ws.page_margins.right,
            "bottom": ws.page_margins.bottom,
            "left": ws.page_margins.left
        }
    }

def extract_column_widths(ws):
    """提取列宽"""
    widths = []
    for col in ws.columns:
        col_letter = get_column_letter(col[0].column)
        widths.append(ws.column_dimensions[col_letter].width)
    return widths

def extract_rows_data(ws):
    """提取行数据"""
    rows = []
    for row in ws.iter_rows():
        row_data = {
            "row_number": row[0].row,
            "height": ws.row_dimensions[row[0].row].height,
            "cells": []
        }
        for cell in row:
            row_data["cells"].append(extract_cell_info(cell))
        rows.append(row_data)
    return rows
```

def convert_to_luckysheet(config):
    """将JSON配置转换为Luckysheet格式"""
    luckysheet_data = {
        "name": config["template_name"],
        "status": "1",
        "order": "0",
        "celldata": [],
        "config": {
            "row": len(config["rows"]),
            "column": len(config["column_widths"]),
            "defaultColWidth": 80,
            "defaultRowHeight": 24
        }
    }
    
    # 填充单元格数据
    for row in config["rows"]:
        for cell in row["cells"]:
            celldata.append({
                "r": row["row_number"] - 1,
                "c": cell["column_number"] - 1,
                "v": cell["text"],
                "m": cell.get("style", ""),
                "rc": cell.get("rowspan", 1),
                "cc": cell.get("colspan", 1)
            })
    
    return luckysheet_data
```

def fill_template_data(config, query_params):
    """自动填报数据"""
    filled_data = copy.deepcopy(config)
    
    for field_name, mapping in config.get("field_mappings", {}).items():
        # 解析数据源
        if "data_source" in mapping:
            table, column = mapping["data_source"].split(".")
            
            # 根据映射类型执行不同查询
            if mapping.get("aggregate_func"):
                value = execute_aggregate_query(table, column, 
                    mapping["aggregate_func"], query_params)
            else:
                value = execute_select_query(table, column, query_params)
            
            # 填充到对应单元格
            fill_cell_value(filled_data, mapping["row"], mapping["column"], value)
    
    return filled_data
```

def is_meaningless(name):
    """检测是否为无意义字段名"""
    meaningless_patterns = [
        r"^tmp[a-zA-Z0-9]+$",
        r"^Unnamed:\s*\d+$",
        r"^field_\d+$",
        r"^col_\d+$",
        r"^\d{4}-\d{2}-\d{2}$",
        r"^sheet\d+$"
    ]
    
    for pattern in meaningless_patterns:
        if re.match(pattern, name, re.IGNORECASE):
            return True
    return False
```

## 四、Phase 3：数据库与API（第2天下午）

### 4.1 数据库表创建

**任务3.1：创建数据库表**
- 文件：`backend/sql/template_tables.sql`
- 表结构：
  - template_configs（模板配置表）
  - template_field_mappings（字段映射表）
  - template_fill_records（填报记录表）

```sql
CREATE TABLE template_configs (
    id SERIAL PRIMARY KEY,
    template_id VARCHAR(50) UNIQUE NOT NULL,
    template_name VARCHAR(100) NOT NULL,
    template_type VARCHAR(50),
    config_json JSONB NOT NULL,
    original_file_path VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE template_field_mappings (
    id SERIAL PRIMARY KEY,
    template_id VARCHAR(50) REFERENCES template_configs(template_id),
    chinese_name VARCHAR(100) NOT NULL,
    english_name VARCHAR(100),
    row_number INTEGER NOT NULL,
    column_number INTEGER NOT NULL,
    data_table VARCHAR(100),
    data_field VARCHAR(100),
    transform_func VARCHAR(100),
    default_value VARCHAR(255),
    mapping_confidence FLOAT DEFAULT 0.0
);

CREATE TABLE template_fill_records (
    id SERIAL PRIMARY KEY,
    template_id VARCHAR(50) REFERENCES template_configs(template_id),
    fill_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    query_params JSONB,
    filled_data JSONB,
    export_file_path VARCHAR(255),
    operator VARCHAR(100)
);
```

### 8.1 命名规范

| 类型 | 规范 | 示例 |
|------|------|------|
| 文件 | 小写蛇形命名 | `template_import_engine.py` |

- 公共API必须有详细文档

- 使用try-except捕获异常
- 返回统一的错误格式
- 记录错误日志
