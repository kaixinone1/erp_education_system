## 二、100%一致性保证机制

### 问题分析

**之前的问题：**

* 导入模板后预览不一致

* 自动填报后导出结果与预览不一致

* 格式丢失、样式变化

**根本原因：**

* 使用了不同的库处理同一文件

* 中间转换过程丢失信息

* 没有保持原始文件结构

### 解决方案：统一技术栈

#### Excel文件处理

```python
# 统一使用 OpenPyXL
from openpyxl import load_workbook, Workbook
from openpyxl.styles import Font, Fill, Border, Alignment
import pandas as pd

class ExcelTemplateHandler:
    def __init__(self, template_path):
        # 加载模板，保持所有格式
        self.wb = load_workbook(template_path)
        self.ws = self.wb.active
        self.template_path = template_path
    
    def preview(self):
        """预览模板 - 直接返回原始文件"""
        # 不做任何修改，直接返回
        return self.template_path
    
    def import_template(self):
        """导入模板 - 保持100%一致"""
        # 保存原始文件的所有信息
        return {
            'file': self.template_path,
            'sheets': self.wb.sheetnames,
            'cells': self._extract_all_cells(),
            'styles': self._extract_all_styles()
        }
    
    def fill_data(self, data):
        """填充数据 - 保持格式不变"""
        # 只修改单元格内容，不修改格式
        for cell_pos, value in data.items():
            cell = self.ws[cell_pos]
            # 保存原始样式
            original_style = {
                'font': cell.font,
                'fill': cell.fill,
                'border': cell.border,
                'alignment': cell.alignment
            }
            # 修改内容
            cell.value = value
            # 恢复样式
            cell.font = original_style['font']
            cell.fill = original_style['fill']
            cell.border = original_style['border']
            cell.alignment = original_style['alignment']
        
        return self.wb
    
    def export(self, output_path):
        """导出结果 - 100%一致"""
        self.wb.save(output_path)
        return output_path
```

```python
from docx import Document

class WordTemplateHandler:
    def __init__(self, template_path):
        self.doc = Document(template_path)
        self.template_path = template_path
    
    def preview(self):
        """预览模板 - 直接返回原始文件"""
        return self.template_path
    
    def import_template(self):
        """导入模板 - 保持100%一致"""
        return {
            'file': self.template_path,
            'paragraphs': self._extract_paragraphs(),
            'tables': self._extract_tables()
        }
    
    def fill_data(self, data):
        """填充数据 - 保持格式不变"""
        # 填充表格
        for table in self.doc.tables:
            for row in table.rows:
                for cell in row.cells:
                    for key, value in data.items():
                        if key in cell.text:
                            # 保持格式替换内容
                            self._replace_cell_content(cell, key, value)
        
        return self.doc
    
    def export(self, output_path):
        """导出结果 - 100%一致"""
        self.doc.save(output_path)
        return output_path
```

### 数据源表注册机制

```python
class DataSourceRegistry:
    """数据源表注册中心"""
    
    def __init__(self):
        self.tables = {}
        self._register_all_tables()
    
    def _register_all_tables(self):
        """注册所有数据源表"""
        # 注册基础信息表
        for table_name, table_info in DATA_SOURCE_TABLES.items():
            self.tables[table_name] = table_info
        
        # 注册退休相关表
        for table_name, table_info in RETIREMENT_TABLES.items():
            self.tables[table_name] = table_info
        
        # 注册绩效工资表
        for table_name, table_info in PERFORMANCE_TABLES.items():
            self.tables[table_name] = table_info
        
        # 注册字典表
        for table_name, table_info in DICTIONARY_TABLES.items():
            self.tables[table_name] = table_info
        
        # 注册其他业务表
        for table_name, table_info in OTHER_TABLES.items():
            self.tables[table_name] = table_info
        
        # 从数据库动态加载其他表
        self._load_database_tables()
    
    def _load_database_tables(self):
        """从数据库加载所有表"""
        import psycopg2
        conn = psycopg2.connect(**DATABASE_CONFIG)
        cursor = conn.cursor()
        
        cursor.execute("""
            SELECT table_name
            FROM information_schema.tables
            WHERE table_schema = 'public'
            AND table_type = 'BASE TABLE'
        """)
        
        for row in cursor.fetchall():
            table_name = row[0]
            if table_name not in self.tables:
                # 自动注册未在配置中的表
                self.tables[table_name] = {
                    'name': table_name,
                    'description': f'{table_name}表',
                    'fields': self._get_table_fields(table_name),
                    'category': '其他',
                    'is_active': True
                }
        
        cursor.close()
        conn.close()
    
    def get_table(self, table_name):
        """获取表信息"""
        return self.tables.get(table_name)
    
    def get_all_tables(self):
        """获取所有表"""
        return self.tables
    
    def get_tables_by_category(self, category):
        """按类别获取表"""
        return {
            name: info 
            for name, info in self.tables.items() 
            if info.get('category') == category
        }

# 全局注册实例
```
