**修改**：`/templates/preview/{filename}` 接口

**实现**：

```python
from openpyxl import load_workbook
from openpyxl.utils import get_column_letter
import base64
import json

@router.get("/templates/preview/{filename}")
async def preview_template(filename: str):
    """预览模板 - 返回完整元数据"""
    try:
        template_path = os.path.join(TEMPLATE_DIR, filename)
        
        if not os.path.exists(template_path):
            raise HTTPException(status_code=404, detail=f"模板不存在: {filename}")
        
        wb = load_workbook(template_path)
        ws = wb.active
        
        # 读取完整元数据
        metadata = {
            'filename': filename,
            'sheets': wb.sheetnames,
            'active_sheet': wb.sheetnames.index(ws.title),
            'cells': [],
            'styles': {},
            'dimensions': {
                'rows': {},
                'columns': {},
                'max_row': ws.max_row,
                'max_column': ws.max_column
            },
            'merged_cells': [],
            'page_setup': {},
            'page_margins': {}
        }
        
        # 1. 读取所有单元格数据和样式
        for row in ws.iter_rows():
            for cell in row:
                cell_key = f"{cell.column_letter}{cell.row}"
                
                # 单元格数据
                if cell.value is not None:
                    metadata['cells'].append({
                        'r': cell.row - 1,  # Luckysheet从0开始
                        'c': cell.column - 1,
                        'v': {
                            'v': cell.value,  # 原始值
                            'm': str(cell.value),  # 显示值
                        }
                    })
                
                # 样式数据
                style = {}
                
                # 字体
                if cell.font:
                    style['font'] = {
                        'name': cell.font.name,
                        'size': cell.font.size,
                        'bold': cell.font.bold,
                        'italic': cell.font.italic,
                        'color': cell.font.color.rgb if cell.font.color and hasattr(cell.font.color, 'rgb') else None,
                        'underline': cell.font.underline,
                        'strike': cell.font.strike
                    }
                
                # 填充
                if cell.fill and cell.fill.patternType:
                    style['fill'] = {
                        'patternType': cell.fill.patternType,
                        'fgColor': cell.fill.fgColor.rgb if cell.fill.fgColor and hasattr(cell.fill.fgColor, 'rgb') else None,
                        'bgColor': cell.fill.bgColor.rgb if cell.fill.bgColor and hasattr(cell.fill.bgColor, 'rgb') else None
                    }
                
                # 边框
                if cell.border:
                    style['border'] = {}
                    for side in ['left', 'right', 'top', 'bottom']:
                        side_obj = getattr(cell.border, side, None)
                        if side_obj and side_obj.style:
                            style['border'][side] = {
                                'style': side_obj.style,
                                'color': side_obj.color.rgb if side_obj.color and hasattr(side_obj.color, 'rgb') else None
                            }
                
                # 对齐
                if cell.alignment:
                    style['alignment'] = {
                        'horizontal': cell.alignment.horizontal,
                        'vertical': cell.alignment.vertical,
                        'wrapText': cell.alignment.wrapText,
                        'textRotation': cell.alignment.textRotation
                    }
                
                # 数字格式
                if cell.number_format and cell.number_format != 'General':
                    style['numberFormat'] = cell.number_format
                
                if style:
                    metadata['styles'][cell_key] = style
        
        # 2. 读取行高
        for row_idx in range(1, ws.max_row + 1):
            row_dim = ws.row_dimensions[row_idx]
            if row_dim and row_dim.height:
                metadata['dimensions']['rows'][row_idx] = row_dim.height
        
        # 3. 读取列宽
        for col_idx in range(1, ws.max_column + 1):
            col_letter = get_column_letter(col_idx)
            col_dim = ws.column_dimensions.get(col_letter)
            if col_dim and col_dim.width:
                metadata['dimensions']['columns'][col_letter] = col_dim.width
        
        # 4. 读取合并单元格
        if hasattr(ws, 'merged_cells') and ws.merged_cells:
            for merged_range in ws.merged_cells.ranges:
                metadata['merged_cells'].append({
                    'range': str(merged_range),
                    'r': merged_range.min_row - 1,
                    'c': merged_range.min_col - 1,
                    'rs': merged_range.max_row - merged_range.min_row + 1,
                    'cs': merged_range.max_col - merged_range.min_col + 1
                })
        
        # 5. 读取页面设置
        if hasattr(ws, 'page_setup') and ws.page_setup:
            ps = ws.page_setup
            metadata['page_setup'] = {
                'paperSize': ps.paperSize,
                'orientation': ps.orientation,
                'fitToPage': ps.fitToPage,
                'fitToWidth': ps.fitToWidth,
                'fitToHeight': ps.fitToHeight
            }
        
        # 6. 读取页边距
        if hasattr(ws, 'page_margins') and ws.page_margins:
            pm = ws.page_margins
            metadata['page_margins'] = {
                'left': pm.left,
                'right': pm.right,
                'top': pm.top,
                'bottom': pm.bottom,
                'header': pm.header,
                'footer': pm.footer
            }
        
        return {
            "success": True,
            "metadata": metadata
        }
        
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
```

```javascript
async function previewTemplate(row) {
  try {
    ElMessage.info('正在加载模板预览...')
    
    const response = await axios.get(`${API_BASE}/templates/preview/${row.filename}`)
    
    if (response.data.success) {
      previewFilename.value = row.filename
      previewDialogVisible.value = true
      
      await nextTick()
      
      const metadata = response.data.metadata
      
      // 转换为Luckysheet格式
      const sheetData = {
        name: metadata.sheets[metadata.active_sheet],
        color: '',
        index: 0,
        status: 1,
        celldata: metadata.cells,
        row: metadata.dimensions.max_row,
        column: metadata.dimensions.max_column,
        config: {}
      }
      
      // 应用样式
      sheetData.celldata.forEach(cell => {
        const cellKey = String.fromCharCode(65 + cell.c) + (cell.r + 1)
        const style = metadata.styles[cellKey]
        
        if (style) {
          // 字体
          if (style.font) {
            if (style.font.bold) cell.v.bl = 1
            if (style.font.italic) cell.v.it = 1
            if (style.font.size) cell.v.fs = style.font.size
            if (style.font.color) cell.v.fc = '#' + style.font.color.substring(2)
          }
          
          // 填充
          if (style.fill && style.fill.fgColor) {
            cell.v.bg = '#' + style.fill.fgColor.substring(2)
          }
          
          // 对齐
          if (style.alignment) {
            const hAlignMap = { 'left': 1, 'center': 0, 'right': 2 }
            const vAlignMap = { 'top': 1, 'center': 0, 'bottom': 2 }
            if (style.alignment.horizontal) cell.v.ht = hAlignMap[style.alignment.horizontal]
            if (style.alignment.vertical) cell.v.vt = vAlignMap[style.alignment.vertical]
            if (style.alignment.wrapText) cell.v.tb = 2
          }
        }
      })
      
      // 应用合并单元格
      if (metadata.merged_cells.length > 0) {
        sheetData.config.merge = {}
        metadata.merged_cells.forEach(merge => {
          const key = `${merge.r}_${merge.c}`
          sheetData.config.merge[key] = {
            r: merge.r,
            c: merge.c,
            rs: merge.rs,
            cs: merge.cs
          }
        })
      }
      
      // 应用行高列宽
      sheetData.rowlen = metadata.dimensions.rows
      sheetData.columnlen = {}
      Object.keys(metadata.dimensions.columns).forEach(col => {
        const colIndex = col.charCodeAt(0) - 65
        sheetData.columnlen[colIndex] = metadata.dimensions.columns[col] * 8
      })
      
      // 清空容器
      const container = document.getElementById('luckysheet-preview')
      if (container) {
        container.innerHTML = ''
      }
      
      // 加载Luckysheet
      await loadLuckysheet()
      
      // 创建预览
      window.luckysheet.create({
        container: 'luckysheet-preview',
        data: [sheetData],
        showtoolbar: false,
        showinfobar: false,
        showsheetbar: false,
        showstatisticBar: false,
        enableAddRow: false,
        enableAddCol: false,
        userInfo: false,
        allowEdit: false
      })
      
      ElMessage.success('模板预览加载成功')
    } else {
      ElMessage.error('预览失败: ' + (response.data.message || '未知错误'))
    }
  } catch (error) {
    console.error('预览错误:', error)
    ElMessage.error('预览失败: ' + error.message)
  }
}

async function loadLuckysheet() {
  if (window.luckysheet) return
  
  // 加载CSS
  const link1 = document.createElement('link')
  link1.rel = 'stylesheet'
  link1.href = 'https://cdn.jsdelivr.net/npm/luckysheet@2.1.13/dist/plugins/css/pluginsCss.css'
  document.head.appendChild(link1)
  
  const link2 = document.createElement('link')
  link2.rel = 'stylesheet'
  link2.href = 'https://cdn.jsdelivr.net/npm/luckysheet@2.1.13/dist/luckysheet.css'
  document.head.appendChild(link2)
  
  // 加载JS
  await new Promise((resolve) => {
    const script1 = document.createElement('script')
    script1.src = 'https://cdn.jsdelivr.net/npm/luckysheet@2.1.13/dist/plugins/js/plugin.js'
    script1.onload = resolve
    document.head.appendChild(script1)
  })
  
  await new Promise((resolve) => {
    const script2 = document.createElement('script')
    script2.src = 'https://cdn.jsdelivr.net/npm/luckysheet@2.1.13/dist/luckysheet.umd.js'
    script2.onload = resolve
    document.head.appendChild(script2)
  })
}
```

```python
def verify_consistency(original_path, preview_metadata):
    """验证预览元数据与原文件的一致性"""
    wb = load_workbook(original_path)
    ws = wb.active
    
    # 验证维度
    assert ws.max_row == preview_metadata['dimensions']['max_row']
    assert ws.max_column == preview_metadata['dimensions']['max_column']
    
    # 验证单元格数量
    original_cells = sum(1 for row in ws.iter_rows() for cell in row if cell.value is not None)
    assert original_cells == len(preview_metadata['cells'])
    
    # 验证合并单元格
    original_merges = len(list(ws.merged_cells.ranges))
    assert original_merges == len(preview_metadata['merged_cells'])
    
    return True
```
